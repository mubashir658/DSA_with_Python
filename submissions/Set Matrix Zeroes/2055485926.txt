class Solution:
    def setZeroes(self, matrix: List[List[int]]) -> None:
        rows_to_zero=set()
        cols_to_zero=set()
        m,n=len(matrix),len(matrix[0])
        for r in range(m):
            for c in range(n):
                if matrix[r][c]==0:
                    rows_to_zero.add(r)
                    cols_to_zero.add(c)
        for r in range(m):
            for c in range(n):
                if r in rows_to_zero or c in cols_to_zero:
                    matrix[r][c]=0
         


        """
        Do not return anything, modify matrix in-place instead.
        """
